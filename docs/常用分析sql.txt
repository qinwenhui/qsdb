                * CAST(CASE WHEN COALESCE(label, 0.0) = 1 THEN CAST(2 AS DECIMAL(38,8)) ELSE CAST(1 AS DECIMAL(38,8)) END AS DECIMAL(38,8))
                * COALESCE(limit_time, 0.0) AS DECIMAL(38,8))
                * CAST(ABS(COALESCE(price, 0.0)) AS DECIMAL(38,8)) * 0.2 AS DECIMAL(38,8)) AS cx_fee,

           CAST((COALESCE(limit_power, 0.0) - COALESCE(max_power, 0.0))
                * CAST(CASE WHEN COALESCE(label, 0.0) = 1 THEN CAST(2 AS DECIMAL(38,8)) ELSE CAST(1 AS DECIMAL(38,8)) END AS DECIMAL(38,8))
